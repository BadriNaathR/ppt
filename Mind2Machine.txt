const fuY = (i: number): number => {
  if (i === 0) return 10;
  if (i === 1) return 32;
  if (i === 2) return 50;
  return 68;
};

const tuY = (fi: number, ti: number): number => {
  const base = fuY(fi);
  const offsets = [-7, 0, 7];
  return base + offsets[ti];
};

function TravelDot({ x1, y1, x2, y2, color, dur = "1.8s" }: {
  x1: number; y1: number; x2: number; y2: number; color: string; dur?: string;
}) {
  return (
    <circle r={3} fill={color} style={{ filter: `drop-shadow(0 0 5px ${color})` }}>
      <animateMotion dur={dur} repeatCount="indefinite" path={`M${x1} ${y1} L${x2} ${y2}`} />
    </circle>
  );
}

function EdgeLine({ x1, y1, x2, y2, color, live, thick = false }: {
  x1: number; y1: number; x2: number; y2: number;
  color: string; live: boolean; thick?: boolean;
}) {
  return (
    <g>
      <line x1={x1} y1={y1} x2={x2} y2={y2}
        stroke="rgba(255,255,255,0.04)" strokeWidth={1.5} markerEnd="url(#arr)" />
      <motion.line x1={x1} y1={y1} x2={x2} y2={y2}
        stroke={live ? color : "rgba(255,255,255,0.08)"}
        strokeWidth={live ? (thick ? 3 : 2) : 1}
        strokeDasharray="8 14"
        animate={{ strokeDashoffset: [0, -44] }}
        transition={{ duration: 1.6, repeat: Infinity, ease: "linear" }}
        style={{ filter: live ? `drop-shadow(0 0 ${thick ? 8 : 4}px ${color})` : "none" }} />
      {live && <TravelDot x1={x1} y1={y1} x2={x2} y2={y2} color={color} dur={thick ? "1.2s" : "1.8s"} />}
    </g>
  );
}
